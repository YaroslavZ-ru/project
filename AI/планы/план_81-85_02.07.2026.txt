ИЗМЕНЕНИЕ 81: update_all_embeddings — thread safety и partial commit
================================================================================

ЧТО НУЖНО СДЕЛАТЬ
-----------------
Привести update_all_embeddings() в соответствии с разделом 32.2
финального описания: commit() после каждого понятия (partial save),
инвалидация кэша под _matrix_lock, обработка ошибок.

ПОЧЕМУ ЭТО НУЖНО
----------------
Сейчас при ошибке одного понятия откатываются ВСЕ изменения (один
большой commit). Partial commit позволяет сохранить успешно
обработанные понятия и продолжить с ошибочными.

ДЕТАЛИ РЕАЛИЗАЦИИ
-------------------
src/knowledge_base.py — update_all_embeddings():

  ИЛЛЮСТРАЦИЯ:
    def update_all_embeddings(self) -> int:
        """Пересчитать эмбеддинги всех понятий.

        commit() после каждого понятия (partial save).
        При ошибке одного — лог ERROR и продолжение.
        В конце — инвалидация кэша под _matrix_lock.

        Returns:
            Количество успешно обновлённых понятий.
        """
        if self._embedding_model is None:
            raise RuntimeError("эмбеддинг-модель не задана")

        rows = self._conn.execute("SELECT id, term FROM concepts").fetchall()
        updated = 0
        for row in rows:
            try:
                blob = self._vector_to_blob(
                    self.compute_concept_embedding(row["term"])
                )
                self._conn.execute(
                    "UPDATE concepts SET embedding=? WHERE id=?",
                    (blob, row["id"])
                )
                self._conn.commit()  # partial save после каждого
                updated += 1
            except Exception as exc:
                self.logger.error(
                    "Ошибка обновления эмбеддинга для %s: %s",
                    row["id"], exc
                )
                self._conn.rollback()  # откат только этого понятия
                continue

        # Инвалидация кэша под _matrix_lock
        with self._matrix_lock:
            self._concepts_cache = None
            self._embeddings_matrix = None
            self._matrix_concept_ids = None

        self.logger.info("Пересчитано эмбеддингов: %d/%d", updated, len(rows))
        return updated

  ВАЖНО:
    - self._conn используется тот же (check_same_thread=False).
    - rollback() откатывает только последнюю транзакцию.
    - _matrix_lock захватывается ТОЛЬКО для инвалидации в конце.

scripts/evaluate.py — алгоритм:
  ИЛЛЮСТРАЦИЯ:
    import json
    from pathlib import Path
    from main import run_pipeline, _init_components
    from src.config import Config

    def evaluate(config_path: str, dataset_path: str) -> dict:
        cfg = Config.from_json(config_path)
        components = _init_components(cfg)
        dataset = json.loads(Path(dataset_path).read_text(encoding="utf-8"))

        total = len(dataset)
        precision_sum = 0.0
        domain_correct = 0

        for item in dataset:
            result = run_pipeline(
                term=item["term"],
                hints=item.get("hints", []),
                debug=False,
                min_confidence=None,
                cfg=cfg,
                *components
            )

            # Context Accuracy
            domain = result.get("selected_context", {}).get("domain", "")
            if domain == item.get("expected_domain"):
                domain_correct += 1

            # Precision@5
            predicted_params = [p["name"] for p in result.get("parameters", [])[:5]]
            expected = set(item.get("expected_params", []))
            hits = sum(1 for p in predicted_params if p in expected)
            precision_sum += hits / 5.0

        context_accuracy = domain_correct / total if total > 0 else 0.0
        avg_precision = precision_sum / total if total > 0 else 0.0

        return {
            "total_cases": total,
            "precision_at_5": round(avg_precision, 4),
            "context_accuracy": round(context_accuracy, 4),
            "domain_correct": domain_correct,
            "domain_incorrect": total - domain_correct,
        }

  Целевые показатели (из раздела 26):
    Precision@5 >= 0.7
    Context Accuracy >= 0.7

    import json
    import sys
    from pathlib import Path

    _RUWORDNET_AVAILABLE = True
    try:
        from ruwordnet import RuWordNet
    except ImportError:
        _RUWORDNET_AVAILABLE = False

    def build_synonyms(output_path: Path) -> dict:
        if not _RUWORDNET_AVAILABLE:
            print("Ошибка: ruwordnet не установлен. pip install ruwordnet")
            sys.exit(1)

        wn = RuWordNet()
        synonyms_dict = {}

        for synset in wn.get_synsets():
            # Лемматизировать название синсета
            lemma = lemmatizer.lemmatize_word(synset.name.lower())
            if not lemma:
                continue

            # Синонимы из синсета (weight=0.8)
            for word in synset.words:
                word_lemma = lemmatizer.lemmatize_word(word.name.lower())
                if word_lemma and word_lemma != lemma:
                    if lemma not in synonyms_dict:
                        synonyms_dict[lemma] = []
                    synonyms_dict[lemma].append({
                        "word": word_lemma,
                        "weight": 0.8
                    })

            # Гиперонимы (weight=0.5)
            for hyper in synset.hypernyms:
                hyper_lemma = lemmatizer.lemmatize_word(hyper.name.lower())
                if hyper_lemma and hyper_lemma != lemma:
                    if lemma not in synonyms_dict:
                        synonyms_dict[lemma] = []
                    synonyms_dict[lemma].append({
                        "word": hyper_lemma,
                        "weight": 0.5
                    })

        # Дедупликация
        for lemma in synonyms_dict:
            seen = set()
            unique = []
            for s in synonyms_dict[lemma]:
                if s["word"] not in seen:
                    seen.add(s["word"])
                    unique.append(s)
            synonyms_dict[lemma] = unique

        output_path.parent.mkdir(parents=True, exist_ok=True)
        output_path.write_text(
            json.dumps(synonyms_dict, ensure_ascii=False, indent=2),
            encoding="utf-8"
        )
        return {"synonyms_count": len(synonyms_dict), "saved_to": str(output_path)}

ИЗМЕНЕНИЕ 85: SearchCache — потокобезопасность через threading.Lock
================================================================================

ЧТО НУЖНО СДЕЛАТЬ
-----------------
Убедиться что кэш результатов поиска в KnowledgeBase использует
threading.Lock для всех операций. Финальное описание (раздел 32.8).

ПОЧЕМУ ЭТО НУЖНО
----------------
При параллельных FastAPI-запросах (asyncio.to_thread) одновременные
записи/чтения OrderedDict без лока приводят к race condition.

ДЕТАЛИ РЕАЛИЗАЦИИ
-------------------
src/knowledge_base.py — кэш поиска:

  ИЛЛЮСТРАЦИЯ:
    from collections import OrderedDict
    import threading
    import time

    class SearchCache:
        def __init__(self, maxsize: int, ttl_seconds: int = 300):
            self._cache: OrderedDict = OrderedDict()
            self._maxsize = maxsize
            self._ttl = ttl_seconds
            self._lock = threading.Lock()

        def get(self, key: tuple) -> list | None:
            with self._lock:
                entry = self._cache.get(key)
                if entry is None:
                    return None
                result, ts = entry
                # Проверка TTL
                if self._ttl > 0 and (time.time() - ts) > self._ttl:
                    del self._cache[key]
                    return None
                # LRU: перемещаем в конец
                self._cache.move_to_end(key)
                return result

        def put(self, key: tuple, result: list) -> None:
            with self._lock:
                if len(self._cache) >= self._maxsize:
                    self._cache.popitem(last=False)
                self._cache[key] = (result, time.time())

        def clear(self) -> None:
            with self._lock:
                self._cache.clear()

  Интеграция в KnowledgeBase:
    self._search_cache = SearchCache(
        maxsize=config.search_cache_size,
        ttl_seconds=config.search_cache_ttl_seconds
    )
