ИЗМЕНЕНИЕ 61: Пакетная обработка запросов
================================================================================

КОНТЕКСТ И ЗАДАЧА
-----------------
Сейчас API обрабатывает только одиночные запросы. Для интеграции с внешними
системами (ETL-пайплайны, батчевая аналитика) нужна возможность передавать
несколько терминов за один HTTP-запрос. Финальное описание (раздел 28)
описывает эндпоинт POST /process_batch.

ПОЧЕМУ ЭТО ВАЖНО
----------------
  - снижает накладные расходы HTTP при обработке множества терминов;
  - удобно для пакетного обновления/проверки базы знаний;
  - позволяет внешним системам отправлять пачки запросов одной транзакцией.

        async def process_batch(request: Request, body: BatchQueryRequest):
            _verify_api_key(request)
            if _cfg is None:
                raise HTTPException(503, detail="Сервис запускается")

            effective_limit = body.max_batch_size or _cfg.max_batch_size
            if len(body.requests) > effective_limit:
                raise HTTPException(
                    400,
                    detail=f"Пакет слишком большой: {len(body.requests)} > {effective_limit}"
                )

            results = []
            for item in body.requests:
                try:
                    result = await asyncio.to_thread(
                        _api_run_pipeline,
                        item.term,
                        [h.strip() for h in item.hints if h.strip()][:3],
                        body.debug,
                        None,  # min_confidence
                        _cfg,
                        _lemmatizer, _synonym_dict, _embedding_model,
                        _vector_cache, _kb, _generative_expander,
                        _session_manager,
                        item.session_id,
                        _metrics,
                    )
                    results.append(QueryResponse.model_validate(result))
                except Exception as exc:
                    logger.error("Ошибка в batch для %r: %s", item.term, exc)
                    results.append(QueryResponse(
                        status="error",
                        term=item.term,
                        warnings=[f"Ошибка обработки: {exc}"],
                    ))

            successful = sum(1 for r in results if r.status == "ok")
            return BatchQueryResponse(
                results=results,
                total=len(results),
                successful=successful,
                failed=len(results) - successful,
            )

ИЗМЕНЕНИЕ 62: Персистентные сессии (SQLite)
================================================================================

КОНТЕКСТ И ЗАДАЧА
-----------------
Сейчас сессии хранятся в памяти (in-memory OrderedDict). При перезапуске
сервера все сессии теряются. Финальное описание (раздел 30) предполагает
опциональное хранение сессий в SQLite для персистентности.

ПОЧЕМУ ЭТО ВАЖНО
----------------
  - сохранение контекста диалога между перезапусками;
  - возможность масштабирования на несколько worker'ов (общая БД);
  - совместимость с Docker (перезапуск контейнера не теряет сессии).

        def __init__(self, config: Config) -> None:
            self._storage = config.session_storage
            self._ttl = config.session_ttl_seconds
            if self._storage == "sqlite":
                self._db_path = Path(config.db_path)
                self._ensure_table()
            else:
                self._data: OrderedDict = OrderedDict()
            self._lock = threading.Lock()

    - Метод _ensure_table():
        if self._storage != "sqlite":
            return
        conn = sqlite3.connect(str(self._db_path))
        conn.execute("PRAGMA journal_mode=WAL")
        conn.execute("""CREATE TABLE IF NOT EXISTS sessions (
            id TEXT PRIMARY KEY, term TEXT NOT NULL, hints TEXT,
            domain TEXT, history TEXT,
            created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
            updated_at DATETIME DEFAULT CURRENT_TIMESTAMP,
            expires_at DATETIME)""")
        conn.execute("CREATE INDEX IF NOT EXISTS idx_sessions_expires ON sessions(expires_at)")
        conn.commit()
        conn.close()

    - Метод _get_conn():
        import sqlite3
        return sqlite3.connect(str(self._db_path), check_same_thread=False)

    - Метод create_session(term, hints) -> str:
        session_id = str(uuid.uuid4())
        now = datetime.datetime.now(datetime.timezone.utc)
        expires = now + datetime.timedelta(seconds=self._ttl)
        if self._storage == "sqlite":
            conn = self._get_conn()
            try:
                conn.execute(
                    "INSERT INTO sessions (id, term, hints, domain, history, expires_at) "
                    "VALUES (?, ?, ?, NULL, '[]', ?)",
                    (session_id, term, json.dumps(hints), expires.isoformat())
                )
                conn.commit()
            finally:
                conn.close()
        else:
            with self._lock:
                self._data[session_id] = {
                    "session_id": session_id, "term": term,
                    "accumulated_hints": list(hints),
                    "selected_domain": None, "interaction_history": [],
                    "created_at": now.isoformat(), "updated_at": now.isoformat(),
                    "ttl_seconds": self._ttl,
                }
        return session_id

    - Метод get_session(session_id) -> dict | None:
        if self._storage == "sqlite":
            conn = self._get_conn()
            try:
                row = conn.execute(
                    "SELECT * FROM sessions WHERE id = ? AND expires_at > datetime('now')",
                    (session_id,)
                ).fetchone()
                if row is None:
                    return None
                return {
                    "session_id": row[0], "term": row[1],
                    "accumulated_hints": json.loads(row[2]) if row[2] else [],
                    "selected_domain": row[3],
                    "interaction_history": json.loads(row[4]) if row[4] else [],
                    "created_at": row[5], "updated_at": row[6],
                }
            finally:
                conn.close()
        else:
            with self._lock:
                session = self._data.get(session_id)
                if session is None:
                    return None
                created = datetime.datetime.fromisoformat(session["created_at"])
                if (datetime.datetime.now(datetime.timezone.utc) - created).total_seconds() > self._ttl:
                    del self._data[session_id]
                    return None
                return session

    - Метод update_session(session_id, new_hints=None, selected_domain=None, response=None):
        if self._storage == "sqlite":
            conn = self._get_conn()
            try:
                if selected_domain is not None:
                    conn.execute("UPDATE sessions SET domain = ?, updated_at = datetime('now') WHERE id = ?",
                                 (selected_domain, session_id))
                if new_hints:
                    row = conn.execute("SELECT hints FROM sessions WHERE id = ?", (session_id,)).fetchone()
                    if row:
                        existing = json.loads(row[0]) if row[0] else []
                        merged = list(dict.fromkeys(existing + new_hints))
                        conn.execute("UPDATE sessions SET hints = ?, updated_at = datetime('now') WHERE id = ?",
                                     (json.dumps(merged), session_id))
                conn.commit()
            finally:
                conn.close()
        else:
            with self._lock:
                session = self._data.get(session_id)
                if session is None:
                    return
                if selected_domain:
                    session["selected_domain"] = selected_domain
                if new_hints:
                    existing = session.get("accumulated_hints", [])
                    session["accumulated_hints"] = list(dict.fromkeys(existing + new_hints))
                session["updated_at"] = datetime.datetime.now(datetime.timezone.utc).isoformat()

    - Метод cleanup_expired() -> int:
        if self._storage == "sqlite":
            conn = self._get_conn()
            try:
                cursor = conn.execute("DELETE FROM sessions WHERE expires_at <= datetime('now')")
                conn.commit()
                return cursor.rowcount
            finally:
                conn.close()
        else:
            with self._lock:
                now = datetime.datetime.now(datetime.timezone.utc)
                expired = [k for k, v in self._data.items()
                           if (now - datetime.datetime.fromisoformat(v["created_at"])).total_seconds() > self._ttl]
                for k in expired:
                    del self._data[k]
                return len(expired)

    - Метод delete_session(session_id):
        if self._storage == "sqlite":
            conn = self._get_conn()
            try:
                conn.execute("DELETE FROM sessions WHERE id = ?", (session_id,))
                conn.commit()
            finally:
                conn.close()
        else:
            with self._lock:
                self._data.pop(session_id, None)

ИЗМЕНЕНИЕ 63: Детекция несвязных подсказок
================================================================================

КОНТЕКСТ И ЗАДАЧА
-----------------
Если пользователь передаёт подсказки, не связанные друг с другом семантически
(например "техника" и "рецепт борща"), система должна предупредить об этом.
Финальное описание (раздел 13) описывает алгоритм: вычислить косинусное
сходство между векторами подсказок и сравнить со порогом.

ПОЧЕМУ ЭТО ВАЖНО
----------------
  - помогает пользователю понять, что подсказки противоречат друг другу;
  - снижает количество некачественных результатов;
  - повышает доверие к системе.

ДЕТАЛИ РЕАЛИЗАЦИИ
-------------------

  src/config.py
    - Добавить поле:
        hints_coherence_threshold: float = 0.2
    - Валидация: float в [0.0, 1.0]

  src/aggregation.py
    - Добавить функцию check_hints_coherence(
          hints: list[str],
          embedding_model: FastTextWrapper,
          threshold: float = 0.2
      ) -> dict:
        1. Если len(hints) < 2: вернуть {"coherent": True, "reason": "недостаточно подсказок"}.
        2. Для каждой пары подсказок (i, j) где i < j:
            vec_i = embedding_model.get_phrase_vector(hints[i])
            vec_j = embedding_model.get_phrase_vector(hints[j])
            norm_i = np.linalg.norm(vec_i)
            norm_j = np.linalg.norm(vec_j)
            if norm_i > 1e-9 and norm_j > 1e-9:
                sim = float(np.dot(vec_i, vec_j) / (norm_i * norm_j))
            else:
                sim = 0.0
            pairs.append({"hint_i": hints[i], "hint_j": hints[j], "similarity": sim})
        3. avg_sim = sum(p["similarity"] for p in pairs) / len(pairs)
        4. Если avg_sim < threshold:
            return {
                "coherent": False,
                "avg_similarity": avg_sim,
                "pairs": pairs,
                "reason": f"Среднее сходство подсказок ({avg_sim:.2f}) ниже порога ({threshold})"
            }
        5. Иначе: вернуть {"coherent": True, "avg_similarity": avg_sim, "pairs": pairs}

ИЗМЕНЕНИЕ 65: Структурированное JSON-lines логирование
================================================================================

КОНТЕКСТ И ЗАДАЧА
-----------------
Сейчас логи форматируются как простой текст: "[2026-07-02 12:00:00] INFO ...".
Финальное описание (раздел 27.3) требует JSON-lines формат для машинного
парсинга: каждая строка — валидный JSON-объект с полями ts, level, logger, msg
и дополнительными контекстными полями.

ПОЧЕМУ ЭТО ВАЖНО
----------------
  -allows централизованный сбор логов (ELK, Loki, CloudWatch);
  - упрощает фильтрацию и поиск по логам;
  - структурированные данные便于 анализ и алертинг.

ДЕТАЛИ РЕАЛИЗАЦИИ
-------------------

  src/utils.py
    - Добавить класс JSONFormatter(logging.Formatter):
        class JSONFormatter(logging.Formatter):
            """Форматтер: каждая строка лога — валидный JSON."""

            def format(self, record: logging.LogRecord) -> str:
                log_data = {
                    "ts": self.formatTime(record, "%Y-%m-%dT%H:%M:%S.%fZ"),
                    "level": record.levelname,
                    "logger": record.name,
                    "msg": record.getMessage(),
                }
                # Добавляем дополнительные поля из record
                for key in ("term", "hints", "elapsed_ms", "n_concepts",
                            "request_id", "status", "concept_id", "path"):
                    val = getattr(record, key, None)
                    if val is not None:
                        log_data[key] = val
                # Exc_info
                if record.exc_info and record.exc_info[0]:
                    log_data["exception"] = self.formatException(record.exc_info)
                return json.dumps(log_data, ensure_ascii=False)
