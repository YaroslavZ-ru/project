      import time
      from main import run_pipeline, _init_components

      def run_benchmark(config_path: str, n: int = 10) -> dict:
          """Запуск бенчмарка и проверка порогов производительности.

          Args:
            config_path: Путь к config.json.
            n: Количество запросов для усреднения.

          Returns:
            Словарь с результатами и статусом прохождения порогов.
          """
          cfg = Config.from_json(Path(config_path))
          components = _init_components(cfg)

          test_queries = [
              {"term": "ключ", "hints": ["техника"]},
              {"term": "ключ гаечный", "hints": ["инструмент"]},
              {"term": "ключ скрипичный", "hints": ["музыка"]},
              {"term": "bolt", "hints": []},
              {"term": "резистор", "hints": ["электроника", "ток"]},
          ]

          timings = {
              "preprocess_ms": [],
              "vectorize_ms": [],
              "search_ms": [],
              "aggregation_ms": [],
              "total_ms": [],
          }

          for _ in range(n):
              for q in test_queries:
                  t0 = time.perf_counter()
                  result = run_pipeline(
                      term=q["term"],
                      hints=q.get("hints", []),
                      debug=False,
                      min_confidence=None,
                      cfg=cfg,
                      *components
                  )
                  total = (time.perf_counter() - t0) * 1000
                  timings["total_ms"].append(total)

          # Вычисление статистики
          stats = {}
          for key, values in timings.items():
              if values:
                  stats[key] = {
                      "mean_ms": round(sum(values) / len(values), 2),
                      "p50_ms": round(sorted(values)[len(values)//2], 2),
                      "p95_ms": round(sorted(values)[int(len(values)*0.95)], 2),
                      "p99_ms": round(sorted(values)[int(len(values)*0.99)], 2),
                  }

          # Проверка порогов (из раздела 25.1)
          thresholds = {
              "total_ms": 300,      # весь pipeline < 300 мс
          }
          checks = {}
          for metric, threshold in thresholds.items():
              if metric in stats:
                  passed = stats[metric]["mean_ms"] < threshold
                  checks[metric] = {
                      "threshold_ms": threshold,
                      "actual_ms": stats[metric]["mean_ms"],
                      "passed": passed,
                  }

          return {
              "stats": stats,
              "checks": checks,
              "all_passed": all(c["passed"] for c in checks.values()),
          }

Описание (раздел 7.1) детально описывает standalone-режим main.py:

  - удобно для скриптов и автоматизации;
  - не требует запуска FastAPI-сервера.

          parser.add_argument("--config", default="configs/config.json")
          parser.add_argument("--debug", action="store_true")
          parser.add_argument("--once", action="store_true")
          args = parser.parse_args()

          cfg = Config.from_json(Path(args.config))
          components = _init_components(cfg)

          # Warm-up модели
          _ = embedding_model.get_word_vector('а')

          if args.once:
              # Однократный режим
              line = sys.stdin.readline().strip()
              if not line:
                  sys.exit(0)
              try:
                  data = json.loads(line)
              except json.JSONDecodeError as e:
                  print(json.dumps(
                      {"status": "error", "message": f"Неверный JSON: {e}"},
                      ensure_ascii=False
                  ))
                  sys.exit(1)
              result = run_pipeline(
                  term=data["term"],
                  hints=data.get("hints", []),
                  debug=args.debug or data.get("debug", False),
                  min_confidence=data.get("min_confidence"),
                  cfg=cfg,
                  *components
              )
              print(json.dumps(result, ensure_ascii=False))

          else:
              # Циклический режим
              logger.info("Старт standalone режима. Введите JSON или 'exit'.")
              while True:
                  try:
                      line = sys.stdin.readline()
                  except (EOFError, KeyboardInterrupt):
                      break
                  if not line:
                      break
                  line = line.strip()
                  if not line:
                      continue
                  if line.lower() in ("exit", "quit"):
                      break
                  try:
                      data = json.loads(line)
                  except json.JSONDecodeError as e:
                      print(json.dumps(
                          {"status": "error", "message": f"Неверный JSON: {e}"},
                          ensure_ascii=False
                      ))
                      continue
                  if "term" not in data or not data["term"]:
                      print(json.dumps(
                          {"status": "error", "message": "Поле 'term' обязательно"},
                          ensure_ascii=False
                      ))
                      continue
                  result = run_pipeline(
                      term=data["term"],
                      hints=data.get("hints", []),
                      debug=args.debug or data.get("debug", False),
                      min_confidence=data.get("min_confidence"),
                      cfg=cfg,
                      *components
                  )
                  print(json.dumps(result, ensure_ascii=False))

          """Проверка готовности проекта к релизу.

              "sessions", "metrics", "utils", "api",

              except ImportError as e:
                  import_errors.append(f"{mod}: {e}")
          checks["all_modules_import"] = len(import_errors) == 0

              if not (project_root / "scripts" / f"{s}.py").exists()
          ]
          checks["all_scripts_exist"] = len(missing_scripts) == 0
