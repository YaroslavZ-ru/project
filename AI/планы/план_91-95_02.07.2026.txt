ИЗМЕНЕНИЕ 91: Таблица FEEDBACK в init_db.py
================================================================================

КОНТЕКСТ И ЗАДАЧА
-----------------
Финальное описание (раздел 24.1) ЯВНО указывает:

  "Создаётся в scripts/init_db.py вместе с остальными таблицами."

Однако план 16 (init_db.py) создаёт только таблицы:
  concepts, parameters, relations, metadata

Таблица feedback с индексами отсутствует. Это прямое противоречие
описанию. Без этой таблицы POST /feedback API будет падать
с sqlite3.OperationalError.

ПОЧЕМУ ЭТО ВАЖНО
-----------------
  - API /feedback не может работать без таблицы;
  - обратная связь пользователей не сохраняется;
  - formula feedback-коррекции (раздел 24.3) не имеет данных.

ДЕТАЛИ РЕАЛИЗАЦИИ
-------------------

  scripts/init_db.py — функция init_db():

    Добавить после создания таблицы metadata (перед commit):

    ИЛЛЮСТРАЦИЯ:
      # Таблица feedback (раздел 24.1 описания)
      conn.execute("""
          CREATE TABLE IF NOT EXISTS feedback (
              id          INTEGER PRIMARY KEY AUTOINCREMENT,
              session_id  TEXT,
              concept_id  TEXT,
              term        TEXT NOT NULL,
              rating      INTEGER NOT NULL CHECK(rating BETWEEN 1 AND 5),
              comment     TEXT,
              created_at  DATETIME DEFAULT CURRENT_TIMESTAMP,
              FOREIGN KEY (concept_id) REFERENCES concepts(id) ON DELETE SET NULL
          )
      """)

      # Индексы для feedback
      conn.execute(
          "CREATE INDEX IF NOT EXISTS idx_feedback_term ON feedback(term)"
      )
      conn.execute(
          "CREATE INDEX IF NOT EXISTS idx_feedback_rating ON feedback(rating)"
      )

    ВАЖНЫЕ ДЕТАЛИ:
      - CHECK(rating BETWEEN 1 AND 5) — валидация на уровне БД.
      - FOREIGN KEY ... ON DELETE SET NULL — при удалении concept_id

      - concept_id может быть NULL — пользователь отказался от предложения.
      - session_id может быть NULL — вызов без сессии.
      - Идемпотентность: CREATE TABLE IF NOT EXISTS, повторный запуск безопасен.
      - PRAGMA foreign_keys = ON уже выполняется в начале init_db().

    Обновить комментарий в начале функции:

      Идемпотентна (CREATE IF NOT EXISTS).

ЧАСТЬ B: Feedback-коррекция confidence

  src/aggregation.py — после aggregate_parameters():

    Добавить функцию apply_feedback_correction(
        candidates: list[dict],
        kb,  # KnowledgeBase для доступа к БД feedback
        config
    ) -> list[dict]:

    ИЛЛЮСТРАЦИЯ:
      def apply_feedback_correction(
          candidates: list[dict],
          kb,
          config
      ) -> list[dict]:
          """Корректировка confidence по обратной связи (раздел 24.3).

          Args:
            candidates: Список кандидатов из search_similar.
            kb: KnowledgeBase для доступа к таблице feedback.
            config: Конфигурация.

          Returns:
            Список кандидатов с обновлённым similarity.
          """
          if not getattr(config, "use_feedback_correction", False):
              return candidates

          feedback_weight = getattr(config, "feedback_weight", 0.1)
          min_votes = getattr(config, "feedback_min_votes", 3)

          for c in candidates:
              concept_id = c.get("concept_id")
              if not concept_id:
                  continue

              # Получить средний рейтинг
              avg_rating = kb.get_avg_rating(concept_id)
              if avg_rating is None:
                  continue  # нет оценок

              votes = kb.get_rating_count(concept_id)
              if votes < min_votes:
                  continue  # статистически ненадёжно

              # Формула из раздела 24.3
              multiplier = 1 + (avg_rating - 3) * feedback_weight
              c["similarity"] = min(c["similarity"] * multiplier, 1.0)

          # Пересортировать по обновлённому similarity
          candidates.sort(key=lambda x: x["similarity"], reverse=True)
          return candidates

  src/knowledge_base.py — два новых метода:

    get_avg_rating(concept_id: str) -> float | None:
      """Средний рейтинг понятия из таблицы feedback.

      Returns:
        Средний рейтинг или None если оценок нет.
      """
      row = self._conn.execute(
          "SELECT AVG(rating) FROM feedback WHERE concept_id = ?",
          (concept_id,)
      ).fetchone()
      if row and row[0] is not None:
          return float(row[0])
      return None

    get_rating_count(concept_id: str) -> int:
      """Количество оценок для понятия."""
      row = self._conn.execute(
          "SELECT COUNT(*) FROM feedback WHERE concept_id = ?",
          (concept_id,)
      ).fetchone()
      return int(row[0]) if row else 0
